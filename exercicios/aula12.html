Triagem de Logs

30 pts · ~35 min

Lê dados/logs_servidor.txt, encontra os IPs que falharam autenticação (linhas FAILED), conta quantas vezes cada um falhou e grava os suspeitos (5 ou mais tentativas) num CSV.

Aulas em jogo: Aula 7 — regex Aula 8 — ficheiros e CSV Aula 9 — dict 

"""
Ficheiro  : ex1_triagem.py
UC        : UC00606
Aula      : 12 — Missão Final
Autor     :
Data      : 18/09/2026
Descrição : Triagem dos logs do SICO-SRV01 — IPs com autenticações falhadas.
Como usar : python ex1_triagem.py
"""
import csv
import re

FICHEIRO_LOG = "dados/logs_servidor.txt"
FICHEIRO_CSV = "suspeitos.csv"
LIMITE = 5

PADRAO_IP = r"___"                 # ← (1) padrão regex de um endereço IPv4


def ler_linhas(caminho):
    """Devolve a lista de linhas do ficheiro (lista vazia se não existir)."""
    try:
        with open(caminho, "r", encoding="utf-8") as f:
            return f.___()             # ← (2) método que devolve TODAS as linhas
    except ___:                        # ← (3) que exceção se o ficheiro não existe?
        print(f"[ERRO] Ficheiro '{caminho}' não encontrado.")
        return []


def contar_falhas(linhas):
    """Conta tentativas FAILED por IP. Devolve dict {ip: nº de falhas}."""
    contagem = {}
    for linha in linhas:
        if linha.startswith("#"):      # ignora o cabeçalho do ficheiro
            continue
        if "___" not in linha:         # ← (4) só as linhas de falha interessam
            continue
        ips = re.___(PADRAO_IP, linha) # ← (5) queres TODOS os IPs da linha
        if not ips:
            continue
        ip = ips[0]
        contagem[ip] = contagem.___(ip, 0) + 1   # ← (6) somar sem dar KeyError
    return contagem


def gravar_csv(suspeitos, caminho):
    """Grava [(ip, tentativas), ...] em CSV, com cabeçalho."""
    with open(caminho, "w", newline="", encoding="utf-8") as f:
        escritor = csv.writer(f)
        escritor.writerow(["ip", "tentativas"])
        escritor.___(suspeitos)        # ← (7) escrever várias linhas de uma vez


def main():
    linhas = ler_linhas(FICHEIRO_LOG)
    if not linhas:
        return

    contagem  = contar_falhas(linhas)
    suspeitos = [(ip, n) for ip, n in contagem.items() if n ___ LIMITE]  # ← (8) 5 ou mais
    suspeitos.sort(key=lambda par: par[1], reverse=True)

    print("=== TRIAGEM DE LOGS — SICO-SRV01 ===")
    print(f"Linhas analisadas : {len(linhas)}")
    print(f"IPs com falhas    : {len(contagem)}")
    print(f"IPs suspeitos     : {len(suspeitos)} (>= {LIMITE} tentativas)\n")

    for ip, n in suspeitos:
        print(f"  {ip:16} {n:3} tentativas falhadas")

    gravar_csv(suspeitos, FICHEIRO_CSV)
    print(f"\nFicheiro '{FICHEIRO_CSV}' gravado com {len(suspeitos)} registos.")


if __name__ == "__main__":
    main()
💡 Pistas:
• Um IPv4 são quatro grupos de 1 a 3 dígitos separados por pontos — atenção que o ponto, em regex, precisa de \.
• dicionario.get(chave, 0) devolve 0 quando a chave ainda não existe.
• O LIMITE = 5 significa 5 ou mais, não mais de 5.
